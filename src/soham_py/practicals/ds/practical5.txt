
import pandas as pd
import numpy as np
from sklearn.preprocessing import MinMaxScaler

# 1. Import Required Libraries
print("Libraries Imported Successfully!")


# 2. Load Open Source Dataset
df = pd.read_csv("data.csv")

print("\n--- Dataset Loaded Successfully ---")
print(df.head())


# 3. Dataset Description and Source
print("\n--- Dataset Description ---")
print("Dataset Name: Student Academic Performance Dataset")
print("Source: Supplied data.csv file")
print("Description: Contains student details, demographic information,")
print("previous education, study hours and academic marks.")


# 4. Check Dimensions of Dataset
print("\n--- Dataset Dimensions ---")
print("Rows:", df.shape[0])
print("Columns:", df.shape[1])
print("Shape:", df.shape)


# 5. Check Missing Values
print("\n--- Missing Values ---")
print(df.isnull().sum())

print("\nTotal Missing Values:")
print(df.isnull().sum().sum())


# 6. Statistical Summary
print("\n--- Statistical Summary ---")
print(df.describe())

print("\n--- Complete Statistical Summary ---")
print(df.describe(include="all"))


# 7. Dataset Information and Data Types
print("\n--- Dataset Information ---")
df.info()

print("\n--- Data Types ---")
print(df.dtypes)


# 8. Variable Descriptions
print("\n--- Variable Descriptions ---")

variable_description = {
    "fNAME": "First name of student",
    "lNAME": "Last name of student",
    "Age": "Age of student",
    "gender": "Gender of student",
    "country": "Country of student",
    "residence": "Student residence",
    "entryEXAM": "Entry examination marks",
    "prevEducation": "Previous education qualification",
    "studyHOURS": "Number of study hours",
    "Python": "Python subject marks",
    "DB": "Database subject marks"
}

for column, description in variable_description.items():
    print(column, ":", description)


# 9. Data Preprocessing - Handle Missing Values
print("\n--- Handling Missing Values ---")

# Fill missing Python marks with median
df["Python"] = df["Python"].fillna(df["Python"].median())

print("Missing Values After Cleaning:")
print(df.isnull().sum())


# 10. Data Formatting and Type Conversion
print("\n--- Data Type Conversion ---")

# Convert categorical columns into category datatype
categorical_columns = [
    "gender", "country", "residence", "prevEducation"
]

for col in categorical_columns:
    df[col] = df[col].astype("category")

# Convert numerical columns into integer datatype
numeric_columns = ["Age", "entryEXAM", "studyHOURS", "Python", "DB"]

for col in numeric_columns:
    df[col] = df[col].astype("int64")

print("\nUpdated Data Types:")
print(df.dtypes)


# 11. Standardize Inconsistent Categorical Values
print("\n--- Standardizing Categorical Values ---")

# Standardize gender
df["gender"] = (
    df["gender"].astype("str").str.strip().str.lower()
    .replace({
        "m": "male",
        "f": "female"
    })
)

# Standardize country names
df["country"] = (
    df["country"].astype("str").str.strip().str.lower()
    .replace({
        "norge": "norway",
        "rsa": "south africa"
    })
)

# Standardize residence
df["residence"] = (
    df["residence"].astype("str").str.strip().str.lower()
    .replace({
        "bi-residence": "bi residence",
        "biresidence": "bi residence",
        "bi_residence": "bi residence"
    })
)

# Standardize previous education
df["prevEducation"] = (
    df["prevEducation"].astype("str").str.strip().str.lower()
    .replace({
        "highschool": "high school",
        "barrrchelors": "bachelors",
        "diploma": "diploma",
        "diplomaaa": "diploma"
    })
)

print(df[categorical_columns].head(10))


# 12. Convert Categorical Variables into Quantitative Variables
print("\n--- Categorical to Quantitative Conversion ---")

df_encoded = pd.get_dummies(
    df,
    columns=categorical_columns,
    drop_first=True,
    dtype=int
)

print(df_encoded.head())

print("\nEncoded Dataset Shape:")
print(df_encoded.shape)


# 13. Data Normalization using MinMaxScaler
print("\n--- Data Normalization ---")

scaler = MinMaxScaler()

numeric_columns = ["Age", "entryEXAM", "studyHOURS", "Python", "DB"]

df_normalized = df_encoded.copy()

df_normalized[numeric_columns] = scaler.fit_transform(
    df_encoded[numeric_columns]
)

print("\nNormalized Dataset:")
print(df_normalized.head())


# 14. Final Dataset Information
print("\n--- Final Dataset Information ---")

print("Original Dataset Shape:", df.shape)
print("Encoded Dataset Shape:", df_encoded.shape)
print("Normalized Dataset Shape:", df_normalized.shape)

print("\nFinal Missing Values:")
print(df_normalized.isnull().sum().sum())

print("\nPractical 5 Executed Successfully!")
