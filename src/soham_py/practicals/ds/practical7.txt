
import pandas as pd
import numpy as np

# 1. Load Dataset
df = pd.read_csv("BigMartData.csv")

print("First 5 Records:")
print(df.head())

print("\nDataset Shape:", df.shape)

print("\nDataset Information:")
df.info()

# 2. Pandas Indexing

# Label-based indexing using loc
print("\nUsing loc:")
print(df.loc[0:4, ["Item_Identifier", "Item_Type", "Item_MRP"]])

# Position-based indexing using iloc
print("\nUsing iloc:")
print(df.iloc[0:5, 0:4])

# Access a single value using at
print("\nUsing at:")
print(df.at[0, "Item_Identifier"])

# Access a single value using iat
print("\nUsing iat:")
print(df.iat[0, 1])

# Boolean indexing
print("\nItems with MRP greater than 200:")
print(df[df["Item_MRP"] > 200].head())

# 3. Identify Missing Values
print("\nMissing Values:")
print(df.isnull().sum())

print("\nTotal Missing Values:")
print(df.isnull().sum().sum())

# 4. Drop Missing Values
df_dropped = df.dropna()

print("\nShape Before Dropping:", df.shape)
print("Shape After Dropping:", df_dropped.shape)

# 5. Fill Missing Values
df_clean = df.copy()

# Fill numerical columns with median
numeric_columns = df_clean.select_dtypes(
    include=np.number
).columns

for col in numeric_columns:
    df_clean[col] = df_clean[col].fillna(
        df_clean[col].median()
    )

# Fill categorical columns with mode
categorical_columns = df_clean.select_dtypes(
    include=["object", "str"]
).columns

for col in categorical_columns:
    if df_clean[col].isnull().any():
        df_clean[col] = df_clean[col].fillna(
            df_clean[col].mode()[0]
        )

print("\nMissing Values After Filling:")
print(df_clean.isnull().sum())

print("\nTotal Missing Values After Filling:",
      df_clean.isnull().sum().sum())

# 6. Merging DataFrames

# Create Record_ID for merging
df_merge = df_clean.reset_index().rename(
    columns={"index": "Record_ID"}
)

df1 = df_merge[
    ["Record_ID", "Item_Identifier", "Item_MRP"]
]

df2 = df_merge[
    ["Record_ID", "Item_Type", "Outlet_Type"]
]

merged_df = pd.merge(
    df1, df2, on="Record_ID"
)

print("\nMerged DataFrame:")
print(merged_df.head())
print("Merged Shape:", merged_df.shape)

# 7. Joining DataFrames
left_df = df_clean[
    ["Item_Identifier", "Item_MRP"]
].copy()

right_df = df_clean[
    ["Outlet_Identifier", "Outlet_Type"]
].copy()

joined_df = left_df.join(right_df)

print("\nJoined DataFrame:")
print(joined_df.head())
print("Joined Shape:", joined_df.shape)

# 8. Concatenating DataFrames

# Row-wise concatenation
df_part1 = df_clean.iloc[:100].copy()
df_part2 = df_clean.iloc[100:200].copy()

concat_rows = pd.concat(
    [df_part1, df_part2],
    axis=0,
    ignore_index=True
)

print("\nRow-wise Concatenation:")
print(concat_rows.head())
print("Concatenated Shape:", concat_rows.shape)

# Column-wise concatenation
df_left = df_clean[
    ["Item_Identifier", "Item_MRP"]
]

df_right = df_clean[
    ["Item_Type", "Outlet_Type"]
]

concat_columns = pd.concat(
    [df_left, df_right],
    axis=1
)

print("\nColumn-wise Concatenation:")
print(concat_columns.head())
print("Concatenated Shape:", concat_columns.shape)

# 9. Final Cleaned Dataset
print("\nFinal Cleaned Dataset:")
print(df_clean.head())

print("\nFinal Dataset Shape:", df_clean.shape)

print("\nTotal Missing Values:",
      df_clean.isnull().sum().sum())

print("\nPractical 7 Executed Successfully!")
