import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.feature_selection import VarianceThreshold, SelectKBest, f_regression, RFE
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.linear_model import LinearRegression
from sklearn.impute import SimpleImputer

# Load dataset
df = pd.read_csv("auto-mpg.csv")

print(df.head())
print("Shape:", df.shape)

# Data cleaning
df['horsepower'] = pd.to_numeric(df['horsepower'], errors='coerce')
df = df.drop(columns=['car name'])

X = df.drop(columns=['mpg'])
y = df['mpg']

imputer = SimpleImputer(strategy='median')
X = pd.DataFrame(imputer.fit_transform(X), columns=X.columns)


# 1. Variance Threshold
vt = VarianceThreshold(threshold=0.0)
X_vt = vt.fit_transform(X)

print("\nVariance Threshold:")
print(X.columns[vt.get_support()].tolist())


# 2. Univariate Feature Selection
ufs = SelectKBest(score_func=f_regression, k=5)
X_ufs = ufs.fit_transform(X, y)

print("\nUnivariate Selection:")
print(X.columns[ufs.get_support()].tolist())


# 3. Recursive Feature Elimination
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

model = LinearRegression()
rfe = RFE(model, n_features_to_select=5)
X_rfe = rfe.fit_transform(X_scaled, y)

print("\nRFE Selected Features:")
print(X.columns[rfe.get_support()].tolist())


# 4. Principal Component Analysis
pca = PCA(n_components=0.95)
X_pca = pca.fit_transform(X_scaled)

print("\nPCA:")
print("Original Features:", X.shape[1])
print("Reduced Features:", X_pca.shape[1])
print("Explained Variance:", sum(pca.explained_variance_ratio_))


# 5. Correlation Analysis
corr = df.corr()

print("\nCorrelation Matrix:")
print(corr)

plt.figure(figsize=(10, 7))
plt.imshow(corr, cmap='coolwarm')
plt.colorbar()
plt.xticks(range(len(corr.columns)), corr.columns, rotation=45)
plt.yticks(range(len(corr.columns)), corr.columns)
plt.title("Correlation Heatmap")
plt.tight_layout()
plt.show()
