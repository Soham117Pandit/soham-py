
import pandas as pd
from scipy import stats

# Load dataset
df = pd.read_csv("Hypothesis_csv1.csv")

print("First 5 Records:")
print(df.head())

print("\nColumns in Dataset:")
print(df.columns.tolist())

# Select numeric column
numeric_cols = df.select_dtypes(include="number").columns

if len(numeric_cols) == 0:
    print("No numeric column found in dataset.")

else:
    # Select lifetime data
    sample = pd.to_numeric(
        df[numeric_cols[0]], errors="coerce"
    ).dropna()

    # Check sample size
    print("\nSample Size:", len(sample))

    if len(sample) != 50:
        print("Warning: Expected 50 observations.")

    # Hypothesis values
    pop_mean = 10000

    print("\nH0: Mean lifetime >= 10000 hours")
    print("H1: Mean lifetime < 10000 hours")

    print("\nSample Mean:", round(sample.mean(), 2))

    # Perform left-tailed one-sample t-test
    t_stat, p_val = stats.ttest_1samp(
        sample,
        popmean=pop_mean,
        alternative="less"
    )

    print("\nT-statistic:", round(t_stat, 4))
    print("P-value:", f"{p_val:.6g}")

    # Test at significance level 0.05
    print("\n--- At 0.05 Significance Level ---")

    if p_val < 0.05:
        print("Reject H0")
        print("Mean lifetime is significantly less than 10000 hours.")
    else:
        print("Fail to Reject H0")
        print("Insufficient evidence that mean lifetime is less than 10000 hours.")

    # Test at significance level 0.01
    print("\n--- At 0.01 Significance Level ---")

    if p_val < 0.01:
        print("Reject H0")
        print("Mean lifetime is significantly less than 10000 hours.")
    else:
        print("Fail to Reject H0")
        print("Insufficient evidence that mean lifetime is less than 10000 hours.")

    print("\nProgram Executed Successfully!")
