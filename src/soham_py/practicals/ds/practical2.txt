
import pandas as pd
from scipy import stats

# Load dataset
licenses = pd.read_csv("federal-firearm-licensees.csv")

print("First 5 Records:")
print(licenses.head())

print("\nColumns in Dataset:")
print(licenses.columns.tolist())

# Count licenses by ZIP code
zip_counts = licenses['PREMISE_ZIP_CODE'].dropna().value_counts()

# Calculate descriptive statistics
mean_zip = zip_counts.mean()

print("\nTotal ZIP Codes:", len(zip_counts))
print("Mean of license counts:", round(mean_zip, 4))
print("Standard Deviation:", round(zip_counts.std(), 4))

# Define hypotheses
pop_mean = 2.75
alpha = 0.05

print("\nH0: Mean license count = 2.75")
print("H1: Mean license count != 2.75")

# Perform one-sample t-test
t_stat, p_val = stats.ttest_1samp(
    zip_counts,
    popmean=pop_mean
)

# Display results
print("\n--- One Sample T-Test Results ---")
print("Hypothesized Mean:", pop_mean)
print("Sample Mean:", round(mean_zip, 4))
print("T-statistic:", round(t_stat, 4))
print("P-value:", round(p_val, 4))
print("Significance Level:", alpha)

# Decision
if p_val < alpha:
    print("\nReject Null Hypothesis (H0)")
    print("Mean license count differs significantly from 2.75.")
else:
    print("\nFail to Reject Null Hypothesis (H0)")
    print("Insufficient evidence that the mean differs from 2.75.")

print("\nProgram Executed Successfully!")
