
import pandas as pd
import numpy as np

from sklearn.impute import SimpleImputer
from sklearn.preprocessing import (
    StandardScaler,
    MinMaxScaler,
    OneHotEncoder,
    KBinsDiscretizer
)

# 1. Load Dataset
df = pd.read_csv("Social_Network_Ads.csv")

print("First 5 Records:")
print(df.head())

print("\nDataset Shape:")
print(df.shape)

# 2. Check Missing Values
print("\nMissing Values:")
print(df.isnull().sum())

# 3. Imputation of Missing Values
numeric_cols = ["Age", "EstimatedSalary"]

imputer = SimpleImputer(strategy="mean")

df[numeric_cols] = imputer.fit_transform(df[numeric_cols])

print("\nMissing Values After Imputation:")
print(df.isnull().sum())

# 4. Encoding using OneHotEncoder
# Convert Purchased into categorical values
df["Purchased_Category"] = df["Purchased"].map({
    0: "Not Purchased",
    1: "Purchased"
})

encoder = OneHotEncoder(
    handle_unknown="ignore",
    sparse_output=False
)

encoded_array = encoder.fit_transform(
    df[["Purchased_Category"]]
)

encoded_df = pd.DataFrame(
    encoded_array,
    columns=encoder.get_feature_names_out(
        ["Purchased_Category"]
    ),
    index=df.index
)

df = pd.concat([df, encoded_df], axis=1)

print("\nAfter Encoding:")
print(df.head())

# 5. Standardization
scaler = StandardScaler()

df[["Age_Standardized", "Salary_Standardized"]] = (
    scaler.fit_transform(
        df[["Age", "EstimatedSalary"]]
    )
)

print("\nAfter Standardization:")
print(df[
    ["Age_Standardized", "Salary_Standardized"]
].head())

# 6. Normalization
normalizer = MinMaxScaler()

df[["Age_Normalized", "Salary_Normalized"]] = (
    normalizer.fit_transform(
        df[["Age", "EstimatedSalary"]]
    )
)

print("\nAfter Normalization:")
print(df[
    ["Age_Normalized", "Salary_Normalized"]
].head())

# 7. Discretization
discretizer = KBinsDiscretizer(
    n_bins=3,
    encode="ordinal",
    strategy="uniform"
)

df["Age_Binned"] = discretizer.fit_transform(
    df[["Age"]]
).astype(int)

print("\nAfter Discretization:")
print(df[["Age", "Age_Binned"]].head(10))

# 8. Final Processed Dataset
print("\nFinal Processed Dataset:")
print(df.head())

print("\nFinal Dataset Shape:")
print(df.shape)

print("\nTotal Missing Values:")
print(df.isnull().sum().sum())

print("\nData Pre-processing Completed Successfully!")
